-- [[ Constructor ]]

-- Builds one triangular tile: 3 vertices/3 edge lines and an invisible collision part that follows the vertex height
function Tile.new(Position: Vector3, TileSize: number, Parent: Instance): ITile_Public
    local self = setmetatable({}, Tile) :: ITile_Private

    self.Position = Position
    self.Vertices = {
        Vector3.new(Position.X + TileSize / 2, Position.Y, Position.Z + TileSize / 2),
        Vector3.new(Position.X - TileSize / 2, Position.Y, Position.Z + TileSize / 2),
        Vector3.new(Position.X, Position.Y, Position.Z - TileSize / 2)
    }
    self.Lines = {}
    self:_CreateLines(Parent)

    self.Part = Instance.new("Part")
    self.Part.Size = Vector3.new(TileSize, 1, TileSize)
    self.Part.Anchored = true
    self.Part.CastShadow = false
    self.Part.Material = Enum.Material.SmoothPlastic
    self.Part.Position = self.Position
    self.Part.Transparency = 1
    self.Part.Parent = Parent

    self:_PlaySFX() -- sound per tile (when appearing)

    return self :: ITile_Public
end

-- [[ Private Methods ]]

-- Draws the triangle 3 edges as thin neon parts
function Tile._CreateLines(self: ITile_Private, Parent: Instance)
    for i = 1, #self.Vertices do
        local StartVertex = self.Vertices[i]
        local EndVertex = self.Vertices[i % #self.Vertices + 1] -- wraps last -- first

        local Line = Instance.new("Part")
        Line.Transparency = 0.35
        Line.CastShadow = false
        Line.Size = Vector3.new(0.1, 0.1, (StartVertex - EndVertex).Magnitude)
        Line.CFrame = CFrame.new((StartVertex + EndVertex) / 2, EndVertex) -- midpoint oriented toward endvertex
        Line.Material = Enum.Material.Neon
        Line.Anchored = true
        Line.CanCollide = false
        Line.Parent = Parent
        table.insert(self.Lines, Line)
    end
end

-- Keeps the collision part height in sync with the animated vertices
function Tile._UpdatePosition(self: ITile_Private)
    local AverageY = (self.Vertices[1].Y + self.Vertices[2].Y + self.Vertices[3].Y) / 3
    self.Part.Position = Vector3.new(self.Position.X, AverageY, self.Position.Z)
end

-- Orients each edge line after vertices move
function Tile._UpdateLines(self: ITile_Private)
    for i = 1, #self.Vertices do
        local StartVertex = self.Vertices[i]
        local EndVertex = self.Vertices[i % #self.Vertices + 1]
        local Line = self.Lines[i]
        Line.CFrame = CFrame.new((StartVertex + EndVertex) / 2, EndVertex)
    end
end

-- Randomized SFX play
function Tile._PlaySFX(self: ITile_Private)
    local Sound = GlassSFX:Clone()
    Sound.Parent = self.Part

    Sound.Ended:Once(function()
        Sound:Destroy()
    end)

    Sound.PlaybackSpeed = math.random(80, 120) / 100
    Sound:Play()
end

-- [[ Public Methods ]]

-- Displaces each vertex Y using a directional sine wave
function Tile.Update(self: ITile_Private, TimeOffset: number, WaveHeight: number, WaveFrequency: number, waveDirection: Vector3)
    for Index, Vertex in ipairs(self.Vertices) do
        local DirectionOffset = (Vertex.X * waveDirection.X + Vertex.Z * waveDirection.Z)
        local NewY = self.Position.Y + math.sin(DirectionOffset * WaveFrequency + TimeOffset) * WaveHeight

        self.Vertices[Index] = Vector3.new(Vertex.X, NewY, Vertex.Z)
    end

    self:_UpdateLines()
    self:_UpdatePosition()
end

-- Cleanup
function Tile.Destroy(self: ITile_Private)
    self.Part:Destroy()

    for _, Line in ipairs(self.Lines) do
        Line:Destroy()
    end

    setmetatable(self, nil)
    table.clear(self :: any)
end

-- Heartbeat <3
function TileController._Runtime(self: PrivateType)
    self.RuntimeConnection = RunService.Heartbeat:Connect(function(Delta: number)
        local Character = LocalPlayer.Character
        if not Character then return end

        local HumanoidRootPart = Character:FindFirstChild("HumanoidRootPart") :: BasePart
        if not HumanoidRootPart then return end

        self:_Update(Delta, HumanoidRootPart)
    end)
end

-- Spawns tiles in AppearDistance
-- Updates existing ones and despawns tiles past DisappearDistance
function TileController._Update(self: PrivateType, Delta: number, HumanoidRootPart: BasePart)
    self.ElapsedTime += Delta
    if self.ElapsedTime < self.UpdateInterval then return end -- throttle

    self.ElapsedTime -= self.UpdateInterval
    self.TimeOffset += self.UpdateInterval * 2 -- advances wave independent of framerate

    local TileSize = self.TileSize + self.TileSpacing
    local Radius = math.ceil(self.AppearDistance / TileSize) -- grid cell radius

    local PX, PZ = HumanoidRootPart.Position.X, HumanoidRootPart.Position.Z
    local CenterX = math.floor(PX / TileSize) * TileSize -- snap player position to grid
    local CenterZ = math.floor(PZ / TileSize) * TileSize

    local InRadiusMap = {}

    for DistanceX = -Radius, Radius do
        for DistanceZ = -Radius, Radius do
            local x = CenterX + DistanceX * TileSize
            local z = CenterZ + DistanceZ * TileSize

            -- Out from boundaries
            if x < self.Boundaries.MinX or x > self.Boundaries.MaxX or
                z < self.Boundaries.MinZ or z > self.Boundaries.MaxZ then
                continue
            end

            -- Set tile status inRadius
            local Key = tostring(x) .. "_" .. tostring(z) -- 10_10 anythik
            InRadiusMap[Key] = true

            -- Create tile
            if not self.TileMap[Key] then
                local Distance = (Vector3.new(x, self.SpawnPosition.Y, z) - HumanoidRootPart.Position).Magnitude
                if Distance <= self.AppearDistance then -- Check distance
                    local TileInstance = Tile.new(
                        Vector3.new(x, self.SpawnPosition.Y, z),
                        self.TileSize,
                        self.Parent
                    )
                    self.TileMap[Key] = TileInstance
                end
            else -- Update tile
                self.TileMap[Key]:Update(
                    self.TimeOffset,
                    self.WaveHeight,
                    self.WaveFrequency,
                    self.WaveDirection
                )
            end
        end
    end

    -- Destroy not inRadius tiles (Optimization)
    for Key, Tile in pairs(self.TileMap) do
        if not InRadiusMap[Key] then
            local distance = (Tile.Position - HumanoidRootPart.Position).Magnitude
            if distance >= self.DisappearDistance then
                Tile:Destroy()
                self.TileMap[Key] = nil
            end
        end
    end
end

import { RunService } from "@rbxts/services"
import { EasingFunction, LinearEase } from "./Easing"
import { IsLerpable, Lerpable, LerpValue } from "./Lerp"

// -----==== Types ====-----

export interface MotionOptions {
  Time: number
  Style?: EasingFunction
  Delay?: number
}

// -----==== Class ====-----

export abstract class Motion {
  // Events
  public OnCompleted?: () => void

  // Instance Elements
  public CurrentValue: Lerpable | undefined
  protected readonly EndValue: Lerpable
  protected readonly StartValue: Lerpable | undefined
  protected readonly Time: number
  protected readonly Delay: number
  protected readonly Style: EasingFunction

  // State
  private Connection?: RBXScriptConnection
  protected CapturedStart: Lerpable | undefined
  protected Elapsed: number = 0
  private IsRunning: boolean = false
  protected IsFinished: boolean = false

  // -----==== Constructor ====-----

  constructor(StartValue: Lerpable | undefined, EndValue: Lerpable, Options: MotionOptions) {
    this.EndValue = EndValue
    this.StartValue = StartValue
    this.Time = Options.Time
    this.Delay = Options.Delay ?? 0
    this.Style = Options.Style ?? LinearEase
  }

  // -----==== Protected Methods ====-----

  protected abstract ApplyValue(Value: Lerpable): void

  protected Step(DeltaTime: number): void {
    this.Elapsed += DeltaTime

    const RawAlpha = math.clamp(this.Elapsed / this.Time, 0, 1)
    const StyledAlpha = this.Style(RawAlpha)
    assert(this.CapturedStart !== undefined, "Motion start value undefined")
    this.CurrentValue = LerpValue(this.CapturedStart, this.EndValue, StyledAlpha)
    this.ApplyValue(this.CurrentValue)

    if (RawAlpha < 1) {
      return
    }

    this.Stop()
    this.IsFinished = true

    if (this.OnCompleted !== undefined) {
      this.OnCompleted()
    }
  }

  private ResetState(): void {
    this.IsFinished = false
    this.Elapsed = -this.Delay
    this.CapturedStart = this.StartValue
  }

  private StartHeartbeat(): void {
    this.IsRunning = true
    this.Connection = RunService.Heartbeat.Connect((DeltaTime: number) => this.Step(DeltaTime))
  }

  // -----==== Public Methods ====-----

  public Play(): void {
    if (this.IsRunning) {
      return
    }

    this.ResetState()
    this.StartHeartbeat()
  }

  public Stop(): void {
    if (this.Connection !== undefined) {
      this.Connection.Disconnect()
      this.Connection = undefined
    }

    this.IsRunning = false
  }

  public Pause(): void {
    this.Stop()
  }

  public Resume(): void {
    if (this.IsFinished || this.IsRunning) {
      return
    }

    if (this.CapturedStart === undefined) {
      this.Play()
      return
    }

    this.StartHeartbeat()
  }

  public IsPlaying(): boolean {
    return this.IsRunning
  }
}

// -----==== InstanceMotion ====-----

export class InstanceMotion extends Motion {
  private readonly Target: Instance
  private readonly Property: string

  constructor(Target: Instance, Property: string, EndValue: Lerpable, Options: MotionOptions) {
    const TargetProperties = Target as unknown as Record<string, unknown>
    const InstanceProperty = TargetProperties[Property]
    assert(IsLerpable(InstanceProperty), "Instance property is not lerpable")
    super(InstanceProperty, EndValue, Options)

    this.Target = Target
    this.Property = Property
  }

  // -----==== Protected Methods ====-----

  protected ApplyValue(Value: Lerpable): void {
    const TargetProperties = this.Target as unknown as Record<string, Lerpable>
    TargetProperties[this.Property] = Value
  }
}

// -----==== ValueMotion ====-----

export class ValueMotion extends Motion {
  public OnStep?: (Value: number) => void

  constructor(StartValue: number, EndValue: number, Options: MotionOptions) {
    super(StartValue, EndValue, Options)
  }

  // -----==== Protected Methods ====-----

  protected ApplyValue(Value: Lerpable): void {
    if (this.OnStep !== undefined) {
      this.OnStep(Value as number)
    }
  }
}

// -----==== Public Functions ====-----

export function Create(Target: Instance, Property: string, EndValue: Lerpable, Options: MotionOptions): InstanceMotion {
  return new InstanceMotion(Target, Property, EndValue, Options)
}

export function CreateValue(StartValue: number, EndValue: number, Options: MotionOptions): ValueMotion {
  return new ValueMotion(StartValue, EndValue, Options)
}

export function AnimateTo(Target: Instance, Property: string, EndValue: Lerpable, Options: MotionOptions): InstanceMotion {
  const MotionInstance = Create(Target, Property, EndValue, Options)
  MotionInstance.Play()
  return MotionInstance
}
